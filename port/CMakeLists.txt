# The web build: the decomp's game sources (upstream/src/game) plus the platform layer in port/src, compiled with
# Emscripten to wasm32.
#   emcmake cmake -S port -B build/web -G Ninja -DCMAKE_BUILD_TYPE=Release
#   cmake --build build/web
# Dependencies come from port/tools/fetch_deps.sh (build/deps).
cmake_minimum_required(VERSION 3.24)
project(thug2web C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# The game has threads of its own (the async loader, music streaming): wasm threads, everything built for them.
add_compile_options(-pthread)
add_link_options(-pthread)

get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(UPSTREAM "${ROOT}/upstream")
set(DEPS "${ROOT}/build/deps")

if(NOT EMSCRIPTEN)
    message(FATAL_ERROR "configure with emcmake (the native 32-bit build comes later)")
endif()

# --- Generated headers upstream's CMake writes -----------------------------------------------------------------------
set(GENERATED "${CMAKE_BINARY_DIR}/generated")
execute_process(COMMAND git rev-parse --short=10 HEAD WORKING_DIRECTORY "${UPSTREAM}"
    OUTPUT_VARIABLE UPSTREAM_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
file(WRITE "${GENERATED}/build_info.h.new"
    "#pragma once\n// Generated by port/CMakeLists.txt.\n#define THUG2_BUILD_COMMIT \"${UPSTREAM_COMMIT} (web)\"\n")
file(COPY_FILE "${GENERATED}/build_info.h.new" "${GENERATED}/build_info.h" ONLY_IF_DIFFERENT)

# --- Libraries ----------------------------------------------------------------------------------------------------
set(IMGUI "${DEPS}/imgui")
add_library(imgui STATIC
    ${IMGUI}/imgui.cpp ${IMGUI}/imgui_draw.cpp ${IMGUI}/imgui_tables.cpp ${IMGUI}/imgui_widgets.cpp
    ${IMGUI}/backends/imgui_impl_sdl2.cpp)
target_include_directories(imgui PUBLIC ${IMGUI})
target_compile_definitions(imgui PUBLIC IMGUI_DISABLE_OBSOLETE_FUNCTIONS IMGUI_DISABLE_DEFAULT_FONT)
target_compile_options(imgui PRIVATE -sUSE_SDL=2 -w)

set(LZHAM "${DEPS}/lzham")
file(GLOB LZHAM_DECOMP_SOURCES ${LZHAM}/lzhamdecomp/*.cpp)
add_library(lzham STATIC ${LZHAM_DECOMP_SOURCES} ${LZHAM}/lzhamlib/lzham_lib.cpp
    ${LZHAM}/lzhamcomp/lzham_lzbase.cpp ${LZHAM}/lzhamcomp/lzham_lzcomp.cpp
    ${LZHAM}/lzhamcomp/lzham_lzcomp_internal.cpp ${LZHAM}/lzhamcomp/lzham_lzcomp_state.cpp
    ${LZHAM}/lzhamcomp/lzham_match_accel.cpp ${LZHAM}/lzhamcomp/lzham_pthreads_threading.cpp)
target_include_directories(lzham PUBLIC ${LZHAM}/include PRIVATE ${LZHAM}/lzhamdecomp ${LZHAM}/lzhamcomp)
target_compile_options(lzham PRIVATE -w)

# --- The game -----------------------------------------------------------------------------------------------------
# Compiled once (thug2_game), linked twice: thug2 for the browser, thug2-node to run headless under Node (the host's
# files, no window, nothing drawn) - for start-up work and tests.
file(GLOB_RECURSE GAME_SOURCES CONFIGURE_DEPENDS "${UPSTREAM}/src/game/*.cpp")
file(GLOB_RECURSE PORT_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp")
# (ImGui's Direct3D 9 backend: the level editor's, on our D3D9)
add_library(thug2_game OBJECT ${GAME_SOURCES} ${PORT_SOURCES} ${IMGUI}/backends/imgui_impl_dx9.cpp)
target_compile_definitions(thug2_game PRIVATE THUG2_STANDALONE THUG2_WEB WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(thug2_game PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}/include"
    "${CMAKE_CURRENT_SOURCE_DIR}/src"
    "${DEPS}/dxvk/include/native/windows"
    "${DEPS}/directx"
    "${UPSTREAM}/src"
    "${GENERATED}"
    "${DEPS}/fonts"
    "${DEPS}/stb"
    "${IMGUI}"
    "${LZHAM}/include")
target_compile_options(thug2_game PRIVATE
    -fms-extensions -fdeclspec -fshort-wchar -include thug2web/msvc_crt.h -sUSE_SDL=2
    -Wno-everything)
target_compile_definitions(thug2_game PRIVATE IMGUI_DISABLE_OBSOLETE_FUNCTIONS IMGUI_DISABLE_DEFAULT_FONT)

set(THUG2_LINK_OPTIONS
    -sUSE_SDL=2
    # (the game uses about 330 MB; a phone's browser may refuse to set aside more than 1 GB for a shared memory)
    -sALLOW_MEMORY_GROWTH -sINITIAL_MEMORY=256MB -sMAXIMUM_MEMORY=1GB -sSTACK_SIZE=4MB
    # The low 8 MB stay free: the exe's .rdata / .data are loaded at their own addresses (0x645000-0x7DD000).
    -sGLOBAL_BASE=8MB
    -sALLOW_TABLE_GROWTH
    # (function names kept: traps and profiles read as the game's functions)
    --profiling-funcs
    -sERROR_ON_UNDEFINED_SYMBOLS=1 -Wl,--error-limit=0)

# The original's linker folded identical functions into one: an empty function (0x005480A0) fills hundreds of vtable
# slots, each called with its own arguments, and the tail jumps pass theirs on untouched - fine on x86 (the caller
# cleans the stack), a trap in wasm (an indirect call checks the exact type). The emulation gives every function in
# the table one type (i64 arguments, converted at each call): every call fits. It costs time on each indirect call,
# so it can be turned off to measure that cost (the game then traps where those calls happen).
option(THUG2_FPCAST_EMU "Emulate x86's function-pointer casts (needed for now)" ON)
if(THUG2_FPCAST_EMU)
    list(APPEND THUG2_LINK_OPTIONS -sEMULATE_FUNCTION_POINTER_CASTS)
    # (and the vtable jumps' forwarders rely on it)
    set_source_files_properties("${CMAKE_CURRENT_SOURCE_DIR}/src/hook/thunks.cpp"
        PROPERTIES COMPILE_DEFINITIONS THUG2_FPCAST_EMU)
endif()

# SDL's video, events and display stood in for: headless under Node (the Emscripten SDL port has no dummy video), and
# in the page, where the game's thread is a worker without the DOM (the page's input comes through a queue instead).
set(SDL_WRAPPED SDL_Init SDL_CreateWindow SDL_GetWindowSize SDL_ShowCursor SDL_PollEvent SDL_GetDesktopDisplayMode
    SDL_SetRelativeMouseMode SDL_GetMouseFocus SDL_GetWindowFromID SDL_SetWindowTitle SDL_NumJoysticks
    SDL_GetMouseState SDL_GetWindowWMInfo SDL_GetKeyboardState)
list(TRANSFORM SDL_WRAPPED PREPEND "-Wl,--wrap=" OUTPUT_VARIABLE SDL_HEADLESS)
# (and in the page, the game controller: the page's pad - touch controls and the browser's gamepads)
set(SDL_PAD_WRAPPED SDL_IsGameController SDL_GameControllerNameForIndex SDL_JoystickNameForIndex
    SDL_GameControllerName SDL_GameControllerOpen SDL_GameControllerFromInstanceID SDL_GameControllerClose
    SDL_GameControllerGetAttached SDL_GameControllerSetPlayerIndex SDL_GameControllerGetPlayerIndex
    SDL_GameControllerGetButton SDL_GameControllerGetAxis SDL_GameControllerGetJoystick SDL_JoystickRumble)
list(TRANSFORM SDL_PAD_WRAPPED PREPEND "-Wl,--wrap=" OUTPUT_VARIABLE SDL_PAD)
set(SDL_PAGE ${SDL_HEADLESS} ${SDL_PAD})

# The page: the game on a worker of its own (PROXY_TO_PTHREAD: it blocks, waits and reads files as on Windows), its
# files fetched as read (port/web/game_files.cpp), its frames drawn by the page's thread with WebGPU.
file(GLOB WEB_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/web/*.cpp")
add_executable(thug2 $<TARGET_OBJECTS:thug2_game> ${WEB_SOURCES})
set_target_properties(thug2 PROPERTIES SUFFIX ".js")
# (the port's own sources outside the game's objects: the same headers and dialect)
set(PORT_INCLUDES
    "${CMAKE_CURRENT_SOURCE_DIR}/include"
    "${CMAKE_CURRENT_SOURCE_DIR}"
    "${CMAKE_CURRENT_SOURCE_DIR}/src"
    "${DEPS}/dxvk/include/native/windows"
    "${DEPS}/directx")
set(PORT_OPTIONS -sUSE_SDL=2 -fms-extensions -fdeclspec -fshort-wchar -include thug2web/msvc_crt.h -Wno-everything)
target_include_directories(thug2 PRIVATE ${PORT_INCLUDES}
    # (WasmFS's own headers: the game folder is a backend of ours)
    "${EMSCRIPTEN_ROOT_PATH}/system/lib/wasmfs")
target_compile_options(thug2 PRIVATE ${PORT_OPTIONS} --use-port=emdawnwebgpu)
target_compile_definitions(thug2 PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(thug2 PRIVATE imgui lzham)
target_link_options(thug2 PRIVATE ${THUG2_LINK_OPTIONS} ${SDL_PAGE}
    -sENVIRONMENT=web,worker -sPROXY_TO_PTHREAD -sWASMFS -sPTHREAD_POOL_SIZE=8 -sEXIT_RUNTIME=0
    # (the page holds the start until THUG2.exe is fetched; its sound reads the mixer's ring in the memory)
    -sEXPORTED_RUNTIME_METHODS=addRunDependency,removeRunDependency,HEAPU8
    --use-port=emdawnwebgpu)

file(GLOB NODE_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/node/*.cpp")
add_executable(thug2-node $<TARGET_OBJECTS:thug2_game> ${NODE_SOURCES})
target_include_directories(thug2-node PRIVATE ${PORT_INCLUDES})
set_target_properties(thug2-node PROPERTIES SUFFIX ".js")
target_compile_options(thug2-node PRIVATE ${PORT_OPTIONS})
target_compile_definitions(thug2-node PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(thug2-node PRIVATE imgui lzham)
target_link_options(thug2-node PRIVATE ${THUG2_LINK_OPTIONS} -sNODERAWFS -sENVIRONMENT=node ${SDL_HEADLESS})

# After each link: every indirect call looks its target up first (thug2_code_slot, port/src/hook/web_hook.cpp) - the
# original's code addresses are function pointers too.
set(ROUTE_CALLS "${CMAKE_CURRENT_SOURCE_DIR}/tools/route_indirect_calls.mjs")
foreach(target thug2 thug2-node)
    add_custom_command(TARGET ${target} POST_BUILD
        COMMAND ${CMAKE_CROSSCOMPILING_EMULATOR} ${ROUTE_CALLS} "$<TARGET_FILE_DIR:${target}>/${target}.wasm"
        VERBATIM)
endforeach()
